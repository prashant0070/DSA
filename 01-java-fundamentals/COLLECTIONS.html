names.add("Ada");

System.out.println(names);        // [Zoe, Ada]
names.remove(Integer.valueOf?);   // careful: remove(1) is INDEX
names.remove("Ada");              // by value

HashMap — word count

Map<String, Integer> freq = new HashMap<>();
for (String w : List.of("to","be","to")) {
  freq.merge(w, 1, Integer::sum);
}
System.out.println(freq.get("to")); // 2

Set — duplicates gone

Set<Integer> ids = new HashSet<>();
ids.add(1); ids.add(1);
System.out.println(ids); // [1]

ArrayDeque — stack & queue

Deque<Integer> stack = new ArrayDeque<>();
stack.push(1); stack.push(2);
stack.pop(); // 2

Deque<String> q = new ArrayDeque<>();
q.offer("A"); q.offer("B");
q.poll(); // A

PriorityQueue

PriorityQueue<Integer> min = new PriorityQueue<>();
min.offer(5); min.offer(1); min.offer(3);
min.peek(); // 1
// for-each is NOT fully sorted — poll() in a loop to drain sorted

Fail-fast CME — fix

List<Integer> list = new ArrayList<>(List.of(1,2,3,4));
// BAD: for (n : list) if (even) list.remove(n);
list.removeIf(n -> n % 2 == 0); // [1, 3]
